import React, { useState } from 'react';
import { useParams } from 'react-router-dom';
import { MapContainer, TileLayer, Marker, useMapEvents } from 'react-leaflet';

function LocationMarker({ setPos }) {
  useMapEvents({ click(e) { setPos(e.latlng); } });
  return null;
}

export default function Booking() {
  const { id } = useParams();
  const [pos, setPos] = useState(null);
  const [form, setForm] = useState({name:'', phone:'', notes:''});

  const send = (e) => {
    e.preventDefault();
    const loc = pos ? \`http://maps.google.com/?q=\${pos.lat},\${pos.lng}\` : 'غير محدد';
    const msg = \`*حجز جديد - سبأ* %0a👤 الاسم: \${form.name}%0a📞 الجوال: \${form.phone}%0a🎫 عرض رقم: \${id}%0a📍 الموقع: \${loc}%0a📝 \${form.notes}\`;
    window.open(\`https://wa.me/966508265296?text=\${msg}\`, '_blank');
  };

  return (
    <div>
      <div className="hero" style={{padding:'10px'}}><h3>📝 إكمال الحجز</h3></div>
      <form onSubmit={send} style={{padding:'15px'}}>
        <label>الاسم:</label>
        <input required style={{width:'100%', padding:'8px', marginBottom:'10px'}} onChange={e=>setForm({...form, name:e.target.value})} />
        
        <label>الجوال (واتساب):</label>
        <input required type="tel" style={{width:'100%', padding:'8px', marginBottom:'10px'}} onChange={e=>setForm({...form, phone:e.target.value})} />

        <label>📍 حدد موقعك:</label>
        <div className="map-box">
          <MapContainer center={[21.54, 39.17]} zoom={12} style={{height:'100%'}}>
            <TileLayer url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png" />
            <LocationMarker setPos={setPos} />
            {pos && <Marker position={pos} />}
          </MapContainer>
        </div>
        
        <button className="btn">إرسال للواتساب </button>
      </form>
    </div>
  );
}
  `
